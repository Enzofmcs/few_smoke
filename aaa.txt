def main(page: ft.Page):
    page.window.title_bar_hidden = True
    page.window.frameless = False
    page.window.full_screen = True

    nav_bar = ft.NavigationBar(
        
        destinations=[
            ft.NavigationBarDestination(icon=ft.icons.HOME, label="Home"),
            ft.NavigationBarDestination(icon=ft.icons.INFO, label="Editor")

        ],
        selected_index=0,
        on_change=lambda e: navigate(e, page)
    )

    def navigate(e, page):
        selected_index = e.control.selected_index
        if selected_index == 0:
            page.go("/")
        elif selected_index == 1:
            page.go("/cart")
        elif selected_index == 2:
            page.go("/editor")

    def route_change(route):
        page.views.clear()
        if route == "/":
            page.views.append(home_view(page))
        elif route == "/cart":
            page.views.append(cart_view())
        elif route == "/editor":
            page.views.append(editor_view())
        page.update()

    page.on_route_change = lambda e: route_change(e.route)

    # Navigate to the initial route
    page.go(page.route)








    ft.Container(
                    ft.Container(
                        content=ft.Image("logo.png", width=200, height=200),
                        alignment=ft.alignment.top_left,

                    ),
                    adaptive=True,
                    height=220,
                    margin=10,
                    border_radius=10,
                    bgcolor=ft.colors.BLUE_100,
                ),
